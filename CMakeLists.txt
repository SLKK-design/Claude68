cmake_minimum_required(VERSION 3.23)
# ── CHANGE THIS for a new project ──────────────────────────────────
project(Claude68)
# ────────────────────────────────────────────────────────────────────

set(CMAKE_C_STANDARD 99)
enable_language(ASM)

SET(CMAKE_C_FLAGS   "-Os")
SET(CMAKE_CXX_FLAGS "-Os -fno-exceptions -fno-rtti")

SET(SSL_ENABLED ON)

include_directories(${CMAKE_CURRENT_SOURCE_DIR}/src)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/src/core)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/src/core/net)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/src/core/helpers)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/src/api)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/src/ui)

# Generate .clangd from template, substituting RETRO68_ROOT so clangd
# can find Classic Mac system headers without hardcoded paths in source.
configure_file(${CMAKE_CURRENT_SOURCE_DIR}/.clangd.in
               ${CMAKE_CURRENT_SOURCE_DIR}/.clangd @ONLY)

add_library(MacHTTP STATIC
        src/core/net/HttpClient.cpp
        src/core/net/HttpResponse.h
        src/core/net/HttpResponse.cpp
        src/core/net/Uri.h
        src/core/net/Uri.cpp
        src/core/net/RootCert.h
        src/core/net/RootCert.cpp
        src/core/net/http_parser.h
        src/core/net/http_parser.c
        src/core/net/yuarel.c
        src/core/net/yuarel.h
        src/core/net/gason.cpp
        src/core/net/gason.hpp
        src/core/net/net_sockets_mac.c
        src/core/net/uECC.c
        # OPT #5: looped Comba multiply, small enough for the 256-byte I-cache
        src/core/net/vli_mult_comba_m68k.s
        # OPT #6: looped Comba square (36 MULU vs 64) — replaces C mul2add path
        src/core/net/vli_square_comba_m68k.s
        # OPT #6b: Solinas P-256 column-sum reduction (generated by
        # HTTPS-68K-Speedtest's tools/gen_solinas_m68k.py) — replaces 10 vli_add/sub RAM passes
        src/core/net/solinas_p256_m68k.s
)

set(UECC_FLAGS "-m68020 -O3 \
 -Dasm_mult=1 \
 -Dasm_square=1 \
 -Dasm_solinas_p256=1 \
 -DuECC_OPTIMIZATION_LEVEL=2 \
 -DuECC_SQUARE_FUNC=1 \
 -DuECC_WORD_SIZE=4 \
 -DuECC_SUPPORTS_secp256r1=1 \
 -DuECC_SUPPORTS_secp160r1=0 \
 -DuECC_SUPPORTS_secp192r1=0 \
 -DuECC_SUPPORTS_secp224r1=0 \
 -DuECC_SUPPORTS_secp256k1=0 \
 -DuECC_VLI_NATIVE_LITTLE_ENDIAN=0")
set_source_files_properties(src/core/net/uECC.c src/core/net/uecc_alt.c PROPERTIES COMPILE_FLAGS "${UECC_FLAGS}")
set_source_files_properties(src/core/net/vli_mult_comba_m68k.s src/core/net/vli_square_comba_m68k.s src/core/net/solinas_p256_m68k.s PROPERTIES COMPILE_FLAGS "-m68020")

IF(${SSL_ENABLED})
    ADD_DEFINITIONS (-DSSL_ENABLED)
    include_directories(${CMAKE_SOURCE_DIR}/src/core/lib/mbedtls/include)
    include_directories(${CMAKE_SOURCE_DIR}/src/core/lib)
    target_link_libraries( MacHTTP ${CMAKE_SOURCE_DIR}/src/core/lib/mbedtls/libmbedtls.a
                                   ${CMAKE_SOURCE_DIR}/src/core/lib/mbedtls/libmbedx509.a
                                   ${CMAKE_SOURCE_DIR}/src/core/lib/mbedtls/libmbedcrypto.a
                                   ${CMAKE_SOURCE_DIR}/src/core/lib/mactcp/libMacTCPHelper.a )
ELSE()
    target_link_libraries( MacHTTP ${CMAKE_SOURCE_DIR}/src/core/lib/mactcp/libMacTCPHelper.a )
ENDIF()

set_target_properties(MacHTTP PROPERTIES COMPILE_OPTIONS -ffunction-sections)

# Application sources: everything in src/ except core/net/ library files
file(GLOB_RECURSE APP_SOURCES
    src/*.cpp
    src/*.c
)
list(FILTER APP_SOURCES EXCLUDE REGEX ".*/src/core/net/.*")
list(FILTER APP_SOURCES EXCLUDE REGEX ".*api_template\\.cpp")
list(APPEND APP_SOURCES
    src/core/net/fast_alloc.c
    # Entropy for the TLS keys: Time Manager jitter (see the file's header)
    src/core/net/mac_entropy.c
    src/core/net/uecc_alt.c
    # OPT #8b: saturated 8x32 X25519 reusing the looped Comba vli asm
    src/core/net/curve25519_sat.c
)

# curve25519_sat: -m68020 for MULU.L in the C glue, asm Comba mult/square
set_source_files_properties(src/core/net/curve25519_sat.c PROPERTIES COMPILE_FLAGS "-m68020 -O3 -Dsat_vli_asm=1")

add_application(Claude68
        ${APP_SOURCES}
        src/wdefshell.r
        CREATOR "Cl68"
)

target_link_libraries( Claude68 MacHTTP )

set_target_properties(Claude68 PROPERTIES
    LINK_FLAGS "-Wl,--allow-multiple-definition -Wl,-gc-sections -Wl,-Map,Claude68.map")

# Touch build_stamp.cpp before every build so __DATE__/__TIME__ reflect actual build time
add_custom_target(build_stamp_touch
    COMMAND ${CMAKE_COMMAND} -E touch ${CMAKE_CURRENT_SOURCE_DIR}/src/core/build_stamp.cpp
    COMMENT "Updating build timestamp"
)
add_dependencies(Claude68 build_stamp_touch)

# Always produce the .APPL as part of default ALL build
add_custom_target(build_appl ALL DEPENDS Claude68_APPL)

# Remove .dsk
add_custom_command(TARGET Claude68_APPL POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E remove -f "${CMAKE_CURRENT_BINARY_DIR}/Claude68.dsk"
)

# Set the Finder flags first, so the copies below carry them
add_custom_command(TARGET Claude68_APPL POST_BUILD
    COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/scripts/set_finder_flags.py" "${CMAKE_CURRENT_BINARY_DIR}" Claude68
    COMMENT "Setting the Finder flags"
)

# Rename to "Claude 68.APPL" (space in name; CMake target must stay Claude68)
add_custom_command(TARGET Claude68_APPL POST_BUILD
    COMMAND cp -p "${CMAKE_CURRENT_BINARY_DIR}/Claude68.APPL"
                  "${CMAKE_CURRENT_BINARY_DIR}/Claude 68.APPL"
    COMMAND cp -p "${CMAKE_CURRENT_BINARY_DIR}/%Claude68.ad"
                  "${CMAKE_CURRENT_BINARY_DIR}/%Claude 68.ad"
    COMMENT "Producing Claude 68.APPL"
)

# Remove spaceless APPL — only "Claude 68.APPL" should exist in the build dir
add_custom_command(TARGET Claude68_APPL POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E remove -f "${CMAKE_CURRENT_BINARY_DIR}/Claude68.APPL"
    COMMENT "Removing Claude68.APPL"
)

install(TARGETS MacHTTP
    DESTINATION "lib"
)
